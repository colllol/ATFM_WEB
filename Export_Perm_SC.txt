PHU LUC A - LENH KIEM TRA NHANH
--------------------------------

Nguon:

SELECT
    (SELECT COUNT(*)
       FROM T_PERMMASTER_SC
      WHERE PERMDATE >= DATE '2026-07-01'
        AND PERMDATE <  DATE '2026-08-01') AS master_count,
    (SELECT COUNT(*)
       FROM T_PERMDETAIL_SC d
      WHERE EXISTS
            (
                SELECT 1
                FROM T_PERMMASTER_SC m
                WHERE m.PERM_ID = d.PERM_ID
                  AND m.PERMDATE >= DATE '2026-07-01'
                  AND m.PERMDATE <  DATE '2026-08-01'
            )) AS detail_count
FROM dual;

SELECT
    (SELECT COUNT(*) FROM ATFM_STAGE.T_PERMMASTER_SC) AS master_count,
    (SELECT COUNT(*) FROM ATFM_STAGE.T_PERMDETAIL_SC) AS detail_count,
    (SELECT COUNT(*)
       FROM ATFM_STAGE.T_PERMDETAIL_SC d
      WHERE NOT EXISTS
            (
                SELECT 1
                FROM ATFM_STAGE.T_PERMMASTER_SC m
                WHERE m.PERM_ID = d.PERM_ID
            )) AS orphan_count
FROM dual;

Dich:

      WHERE PERMDATE >= DATE '2026-07-01'
        AND PERMDATE <  DATE '2026-08-01') AS master_count,
    (SELECT COUNT(*)
       FROM ATFM.T_PERMDETAIL_SC d
      WHERE EXISTS
            (
                SELECT 1
